class DimensionalAnalyzer {
public:
    std::map<std::string, Unit> varUnits;
    std::map<std::string, Unit> funcUnits; // For functions like sin (dimensionless)
    Unit getUnit(const RCP<const Basic>& expr) {
        if (is_a<Symbol>(*expr)) {
            auto it = varUnits.find(down_cast<const Symbol&>(*expr).get_name());
            return it != varUnits.end() ? it->second : Unit{};
        } else if (is_a<Add>(*expr)) {
            auto args = down_cast<const Add&>(*expr).get_args();
            Unit u = getUnit(args[0]);
            for (size_t i = 1; i < args.size(); ++i) {
                if (u != getUnit(args[i])) return Unit{}; // Inconsistent
            }
            return u;
        } else if (is_a<Mul>(*expr)) {
            auto args = down_cast<const Mul&>(*expr).get_args();
            Unit u;
            for (auto& arg : args) u = u * getUnit(arg);
            return u;
        } else if (is_a<Pow>(*expr)) {
            Unit baseU = getUnit(down_cast<const Pow&>(*expr).get_base());
            int exp = eval_double(*down_cast<const Pow&>(*expr).get_exp());
            Unit u = baseU;
            for (int i = 1; i < exp; ++i) u = u * baseU;
            return u;
        } // Add for functions, etc.
        return Unit{}; // Dimensionless
    }
    bool validateEquation(const RCP<const Basic>& lhs, const RCP<const Basic>& rhs) {
        return getUnit(lhs) == getUnit(rhs);
    }
};
