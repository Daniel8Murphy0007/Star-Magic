3D N-Body Simulation Implementation an N-body simulation models the dynamics of a system of particles interacting under gravity, commonly used in astrophysics to simulate star clusters, galaxies, or planetary systems. Below, I'll provide a complete, standalone Python implementation for a basic 3D N-body simulation using direct summation (O(N²) complexity, suitable for small numbers of particles). It uses NumPy for vector operations and Matplotlib for 3D visualization.This code initializes particles with random masses, positions, and velocities, computes gravitational forces, updates positions/velocities using a simple Euler integration method, and plots the trajectories. For larger N or more accuracy, consider optimizations like Barnes-Hut tree algorithms or better integrators (e.g., leapfrog).

import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

# Constants
G = 6.67430e-11  # gravitational constant m^3 kg^-1 s^-2

# Particle class
class Particle:
    def __init__(self, mass, position, velocity):
        self.mass = mass
        self.position = np.array(position, dtype=float)
        self.velocity = np.array(velocity, dtype=float)
        self.force = np.zeros(3)

# Function to compute forces
def compute_forces(particles):
    for i, p1 in enumerate(particles):
        p1.force = np.zeros(3)
        for j, p2 in enumerate(particles):
            if i != j:
                r_vec = p2.position - p1.position
                r = np.linalg.norm(r_vec)
                if r > 0:
                    f_mag = G * p1.mass * p2.mass / r**2
                    p1.force += f_mag * r_vec / r

# Update positions and velocities (Euler method for simplicity)
def update_particles(particles, dt):
    for p in particles:
        acceleration = p.force / p.mass
        p.velocity += acceleration * dt
        p.position += p.velocity * dt

# Simulation parameters
num_particles = 5  # Increase for more particles (but computationally expensive)
dt = 1e4  # time step in seconds (adjust based on scale)
num_steps = 100  # number of simulation steps

# Initialize particles randomly (example: stellar scales)
np.random.seed(42)  # For reproducibility
particles = []
for _ in range(num_particles):
    mass = np.random.uniform(1e26, 1e27)  # masses like small stars (kg)
    position = np.random.uniform(-1e11, 1e11, 3)  # positions within ~1 AU (m)
    velocity = np.random.uniform(-1e3, 1e3, 3)  # velocities ~1 km/s (m/s)
    particles.append(Particle(mass, position, velocity))

# Run simulation and collect positions for plotting
positions_over_time = np.zeros((num_steps, num_particles, 3))
for step in range(num_steps):
    compute_forces(particles)
    update_particles(particles, dt)
    for i, p in enumerate(particles):
        positions_over_time[step, i] = p.position

# Plot the trajectories in 3D
fig = plt.figure(figsize=(10, 8))
ax = fig.add_subplot(111, projection='3d')
for i in range(num_particles):
    ax.plot(positions_over_time[:, i, 0], positions_over_time[:, i, 1], positions_over_time[:, i, 2], label=f'Particle {i+1}')

ax.set_xlabel('X (m)')

ax.set_title('3D N-Body Simulation Trajectories')

// Compute forces
void NBodySimulation::compute_forces() {
    for(size_t i = 0; i < particles.size(); ++i) {
        particles[i].force = {0.0, 0.0, 0.0};
        for(size_t j = 0; j < particles.size(); ++j) {
            if(i != j) {
                std::array<double, 3> r_vec;
                double r_squared = 0.0;
                for(int k = 0; k < 3; ++k) {
                    r_vec[k] = particles[j].position[k] - particles[i].position[k];
                    r_squared += r_vec[k] * r_vec[k];
                }
                double r = std::sqrt(r_squared);
                if(r > 0.0) {
                    double f_mag = G * particles[i].mass * particles[j].mass / r_squared;
                    for(int k = 0; k < 3; ++k) {
                        particles[i].force[k] += f_mag * r_vec[k] / r;
                    }
                }
            }
        }
    }
    /*
    Direct summation: O(N²), vectorized manually with arrays.
    */
}

// Update particles (Euler)
void NBodySimulation::update_particles(double dt) {
    for(auto& p : particles) {
        std::array<double, 3> acceleration;
        for(int k = 0; k < 3; ++k) {
            acceleration[k] = p.force[k] / p.mass;
            p.velocity[k] += acceleration[k] * dt;
            p.position[k] += p.velocity[k] * dt;
        }
    }
    /*
    Euler integration: Simple but low accuracy; for production use leapfrog/Verlet.
    */
}

}

        for(int k = 0; k < 3; ++k) {
            p.position[k] *= expansion_factor;
        }

}

    std::array<double, 3> position;
    std::array<double, 3> velocity;
    std::array<double, 3> force;

    double total_mass;                   // Total mass in this node

    std::unique_ptr<OctreeNode> children[8]; // 8 octants
    int particle_index;                  // -1 if internal node or empty, else particle index

        : center(c), half_size(hs), total_mass(0.0),

        for(int i = 0; i < 8; ++i) children[i] = nullptr;
    }

    void insert(int idx, const std::vector<Particle>& particles);
    void compute_mass_distribution(const std::vector<Particle>& particles);
    void compute_force_on_particle(Particle& p, double theta, double G, double softening) const;
};

/*
=== LEAPFROG INTEGRATOR (Kick-Drift-Kick variant) ===

The leapfrog method is symplectic (preserves phase space volume) and time-reversible,
making it ideal for gravitational N-body simulations with excellent energy conservation.

Algorithm (KDK - Kick-Drift-Kick):
1. Kick: v_{n+1/2} = v_n + a_n * dt/2
2. Drift: x_{n+1} = x_n + v_{n+1/2} * dt  
3. Compute: a_{n+1} from new positions
4. Kick: v_{n+1} = v_{n+1/2} + a_{n+1} * dt/2

Energy conservation: O(dt²) per step, bounded error over long times.
*/

void NBodySimulation::initialize_leapfrog() {
    // Compute initial forces and set up half-step velocities
    if(use_barnes_hut) {
        compute_forces_barnes_hut();
    } else {
        compute_forces();
    }
    
    // First half-kick: v_{1/2} = v_0 + a_0 * dt/2
    for(auto& p : particles) {
        for(int k = 0; k < 3; ++k) {
            double acc = p.force[k] / p.mass;
            p.velocity_half[k] = p.velocity[k] + acc * dt * 0.5;
        }
    }
    leapfrog_initialized = true;
}

void NBodySimulation::update_particles_leapfrog(double dt) {
    if(!leapfrog_initialized) {
        initialize_leapfrog();
    }
    
    // Drift: x_{n+1} = x_n + v_{n+1/2} * dt
    for(auto& p : particles) {
        for(int k = 0; k < 3; ++k) {
            p.position[k] += p.velocity_half[k] * dt;
        }
    }
    
    // Compute new forces at updated positions
    if(use_barnes_hut) {
        compute_forces_barnes_hut();
    } else {
        compute_forces();
    }
    
    // Kick: Update half-step velocity and full velocity
    for(auto& p : particles) {
        for(int k = 0; k < 3; ++k) {
            double acc = p.force[k] / p.mass;
            // v_{n+1} = v_{n+1/2} + a_{n+1} * dt/2 (for output/energy calc)
            p.velocity[k] = p.velocity_half[k] + acc * dt * 0.5;
            // v_{n+3/2} = v_{n+1/2} + a_{n+1} * dt (for next drift)
            p.velocity_half[k] += acc * dt;
        }
    }
}

/*
=== BARNES-HUT TREE ALGORITHM O(N log N) ===

For large N (>1000), direct O(N²) force summation becomes prohibitive.
Barnes-Hut uses an octree to approximate distant particle groups as single masses.

Algorithm:
1. Build octree containing all particles
2. For each particle, traverse tree:
   - If node is far enough (s/d < theta), treat as single mass at center-of-mass
   - Otherwise, recurse into children
   where s = node size, d = distance to node, theta = opening angle (~0.5)

Complexity: O(N log N) typically, O(N²) worst case (unlikely in practice).
*/

int OctreeNode::get_octant(const std::array<double, 3>& pos) const {
    int octant = 0;
    if(pos[0] >= center[0]) octant |= 1;
    if(pos[1] >= center[1]) octant |= 2;
    if(pos[2] >= center[2]) octant |= 4;
    return octant;
}

void OctreeNode::insert(int idx, const std::vector<Particle>& particles) {
    if(is_leaf && particle_index == -1) {
        // Empty leaf: store particle
        particle_index = idx;
        return;
    }
    
    if(is_leaf && particle_index >= 0) {
        // Occupied leaf: subdivide
        is_leaf = false;
        int old_idx = particle_index;
        particle_index = -1;
        
        // Create child for existing particle
        int oct_old = get_octant(particles[old_idx].position);
        double hs = half_size * 0.5;
        std::array<double, 3> child_center;
        child_center[0] = center[0] + ((oct_old & 1) ? hs : -hs);
        child_center[1] = center[1] + ((oct_old & 2) ? hs : -hs);
        child_center[2] = center[2] + ((oct_old & 4) ? hs : -hs);
        children[oct_old] = std::make_unique<OctreeNode>(child_center, hs);
        children[oct_old]->insert(old_idx, particles);
    }
    
    // Insert new particle into appropriate child
    int oct_new = get_octant(particles[idx].position);
    if(!children[oct_new]) {
        double hs = half_size * 0.5;
        std::array<double, 3> child_center;
        child_center[0] = center[0] + ((oct_new & 1) ? hs : -hs);
        child_center[1] = center[1] + ((oct_new & 2) ? hs : -hs);
        child_center[2] = center[2] + ((oct_new & 4) ? hs : -hs);
        children[oct_new] = std::make_unique<OctreeNode>(child_center, hs);
    }
    children[oct_new]->insert(idx, particles);
}

void OctreeNode::compute_mass_distribution(const std::vector<Particle>& particles) {
    if(is_leaf) {
        if(particle_index >= 0) {
            total_mass = particles[particle_index].mass;
            center_of_mass = particles[particle_index].position;
        }
        return;
    }
    
    total_mass = 0.0;
    center_of_mass = {0.0, 0.0, 0.0};
    
    for(int i = 0; i < 8; ++i) {
        if(children[i]) {
            children[i]->compute_mass_distribution(particles);
            double child_mass = children[i]->total_mass;
            total_mass += child_mass;
            for(int k = 0; k < 3; ++k) {
                center_of_mass[k] += child_mass * children[i]->center_of_mass[k];
            }
        }
    }
    
    if(total_mass > 0.0) {
        for(int k = 0; k < 3; ++k) {
            center_of_mass[k] /= total_mass;
        }
    }
}

void OctreeNode::compute_force_on_particle(Particle& p, double theta, double G, double softening) const {
    if(total_mass == 0.0) return;
    
    // Compute distance to center of mass
    std::array<double, 3> r_vec;
    double r_squared = 0.0;
    for(int k = 0; k < 3; ++k) {
        r_vec[k] = center_of_mass[k] - p.position[k];
        r_squared += r_vec[k] * r_vec[k];
    }
    double r = std::sqrt(r_squared + softening * softening);  // Add softening
    
    // Skip if particle is at this node's center of mass (self-interaction)
    if(r < softening) return;
    
    // Barnes-Hut criterion: s/d < theta
    double s = 2.0 * half_size;  // Node size
    if(is_leaf || (s / r < theta)) {
        // Treat as single mass
        double f_mag = G * p.mass * total_mass / (r_squared + softening * softening);
        for(int k = 0; k < 3; ++k) {
            p.force[k] += f_mag * r_vec[k] / r;
        }
    } else {
        // Recurse into children
        for(int i = 0; i < 8; ++i) {
            if(children[i]) {
                children[i]->compute_force_on_particle(p, theta, G, softening);
            }
        }
    }
}

void NBodySimulation::build_octree() {
    // Find bounding box
    double min_coord = -1e12, max_coord = 1e12;
    for(const auto& p : particles) {
        for(int k = 0; k < 3; ++k) {
            min_coord = std::min(min_coord, p.position[k]);
            max_coord = std::max(max_coord, p.position[k]);
        }
    }
    
    double half_size = (max_coord - min_coord) * 0.6;  // Slightly larger
    std::array<double, 3> center = {(min_coord + max_coord) * 0.5, 
                                     (min_coord + max_coord) * 0.5, 
                                     (min_coord + max_coord) * 0.5};
    
    octree_root = std::make_unique<OctreeNode>(center, half_size);
    
    for(size_t i = 0; i < particles.size(); ++i) {
        octree_root->insert(static_cast<int>(i), particles);
    }
    
    octree_root->compute_mass_distribution(particles);
}

void NBodySimulation::compute_forces_barnes_hut() {
    build_octree();
    
    for(auto& p : particles) {
        p.force = {0.0, 0.0, 0.0};
        octree_root->compute_force_on_particle(p, theta, G, softening);
    }
}

/*
=== ENERGY CONSERVATION CHECK ===

Total energy E = T + U should be conserved in a closed gravitational system.
Monitoring ΔE/E₀ validates the integrator's accuracy.

- Kinetic energy: T = Σ (1/2) m_i |v_i|²
- Potential energy: U = -Σ_{i<j} G m_i m_j / r_{ij}
- Energy drift: |E(t) - E(0)| / |E(0)| should be small (< 0.01 for good integrators)
*/

double NBodySimulation::compute_kinetic_energy() const {
    double T = 0.0;
    for(const auto& p : particles) {
        double v_squared = 0.0;
        for(int k = 0; k < 3; ++k) {
            v_squared += p.velocity[k] * p.velocity[k];
        }
        T += 0.5 * p.mass * v_squared;
    }
    return T;
}

double NBodySimulation::compute_potential_energy() const {
    double U = 0.0;
    for(size_t i = 0; i < particles.size(); ++i) {
        for(size_t j = i + 1; j < particles.size(); ++j) {
            double r_squared = 0.0;
            for(int k = 0; k < 3; ++k) {
                double dx = particles[j].position[k] - particles[i].position[k];
                r_squared += dx * dx;
            }
            double r = std::sqrt(r_squared + softening * softening);
            if(r > 0.0) {
                U -= G * particles[i].mass * particles[j].mass / r;
            }
        }
    }
    return U;
}

double NBodySimulation::compute_total_energy() const {
    return compute_kinetic_energy() + compute_potential_energy();
}

void NBodySimulation::check_energy_conservation(int step) {
    double current_energy = compute_total_energy();
    double relative_error = std::abs((current_energy - initial_energy) / initial_energy);
    
    std::cout << "Step " << step << ": E = " << current_energy 
              << ", ΔE/E₀ = " << relative_error;
    
    if(relative_error > 0.01) {
        std::cout << " [WARNING: Energy drift > 1%]";
    } else if(relative_error > 0.001) {
        std::cout << " [Caution: Energy drift > 0.1%]";
    } else {
        std::cout << " [OK]";
    }
    std::cout << std::endl;
}
