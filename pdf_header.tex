% ── Unicode glyph fallbacks for chars absent from DejaVu Serif ──────────────
% DejaVu Sans covers U+2713 (✓); math mode covers ⊙ and \checkmark.
% The super/subscript blocks (U+2070-209F, U+1D00-1DBF) are not reliably
% present in DejaVu Serif — map them to \textsuperscript / \textsubscript.
\usepackage{newunicodechar}
\newfontfamily\symbolfont{DejaVu Sans}
% Astronomical / scientific symbols
\newunicodechar{☉}{\ensuremath{\odot}}          % U+2609 Sun → math odot
\newunicodechar{✓}{{\symbolfont ✓}}             % U+2713 checkmark → DejaVu Sans
\newunicodechar{✗}{{\symbolfont ✗}}             % U+2717 ballot x → DejaVu Sans
% Superscript digits / signs (U+2070-U+207F)
\newunicodechar{⁰}{\textsuperscript{0}}
\newunicodechar{⁴}{\textsuperscript{4}}
\newunicodechar{⁵}{\textsuperscript{5}}

\newunicodechar{⁻}{\textsuperscript{-}}
\newunicodechar{⁺}{\textsuperscript{+}}
